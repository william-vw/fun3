- install macos runtime
(prebuilt not working for arm64)
% cd cpp/grammar/antlr4-cpp-runtime-4.13.2-macos
% ./deploy_macos.sh

- generate lexer, parser
% java -jar ../../grammar/antlr-4.13.2-complete.jar -Dlanguage=Cpp -o ./parser ../../grammar/n3.g4

- compile lexer, parser
% cd parser
% g++ -O3 -std=c++17 -Iantlr4-runtime/ -Lantlr4-cpp-runtime-4.13.2-source/runtime -lantlr4-runtime -Wall -shared parser/*.cpp -o libn3Parser.so

- run n3 parser solo
% g++ -std=c++17 -o parsen3 parse_n3_solo.cpp -Iparser/ -Iantlr4-runtime/ -Lantlr4-cpp-runtime-4.13.2-source/runtime -lantlr4-runtime -L. -ln3Parser
% ./parsen3 /Users/wvw/git/n3/fun3/python/tests-bench/zika/data/gen20000_pt2.n3

- compile n3 parser as library, n3 parser bindings
% ./build.sh

- run in python
% python parse_n3.py

>>> import n3Parser
>>> n3Parser.parse("/Users/wvw/git/n3/fun3/python/tests-bench/zika/data/gen20000_pt2.n3")
